get_property(metal_vasp_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(metal_vasp_target IN LISTS metal_vasp_targets)
  target_compile_options(${metal_vasp_target} PRIVATE
    $<$<COMPILE_LANGUAGE:CXX,OBJCXX>:-fno-fast-math;-ffp-contract=off>)
  target_compile_definitions(${metal_vasp_target} PRIVATE
    $<$<COMPILE_LANGUAGE:CXX,OBJCXX>:METAL_VASP_DEFAULT_PROFILE="${METAL_VASP_PROFILE}">
    $<$<COMPILE_LANGUAGE:CXX,OBJCXX>:METAL_VASP_DEFAULT_LANGUAGE="${METAL_VASP_LANGUAGE_VERSION}">
    $<$<COMPILE_LANGUAGE:CXX,OBJCXX>:METAL_VASP_FP32_FAST_MATH=$<BOOL:${METAL_VASP_FAST_MATH}>>)
endforeach()
